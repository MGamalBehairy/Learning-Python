# Nested Loop   => is A loop within Another Loop    (Outer, Inner)
#                  Outer Loop:
#                        Inner Loop:

for x in range(3):
    for y in range(1, 10):
        print(y, end=" ")   # will print from 1 to 9  at the same line      "  end=""  ", the default is end="\n"
    print()

# print a Rectangle 
rows = int(input("Enter number of rows: "))
columns = int(input("Enter number of columns: "))
symbol = input("Enter the symbol: ")
for x in range(rows):
    for y in range(columns):
        print(symbol, end="")
    print()
