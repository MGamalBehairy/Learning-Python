# program to get the circumference
import math
radius = float(input("Enter the radius of a circle"))
circumference = 2 * math.pi * radius
area = math.pi * pow(radius,2)
print(f"The circumference is: {round(circumference), 2} cm")	# =>  round(x, 2) => get only two float point
print(f"The are of the circle is: {area}cm^2 (alt + 0178)")

# program to calculate the hypotenuse of a right angles triangle

import math
a = float(input("Enter side A: "))
b = float(input("Enter side B: "))

c = math.sqrt(pow(a, 2) + pow(b, 2))
print(f"The hypotenuse of a right angle triangle is: {round(c,2)}")
