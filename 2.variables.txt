----------------------Variables--------------------------------

#Strings
name = "Mohamed"
print(f"Hello {name}")		=> output: Hello Mohamed
=> f => stands for format, u should insert it first to be able to use a variable within text 
print("Hello {name}")	=> output:  Hello {name}

food = "pizza"
print(f"I like {food}")

# integers
age = 25
print(f"You're {age} years old")

#Float is a numeber but it contains a decimal portion
price = 10.99
print(f"The price is {price}")

gpa = 3.2
print(f"Your GPA is: {gpa}")
 
# Boolean
is_student = True
is_teacher = False
print(f"Are you a student?: {is_student}")
