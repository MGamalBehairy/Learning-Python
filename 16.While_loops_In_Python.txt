# While Loop    => Execute some Condition While Some Condition remains True

name = input("Enter your name: ")

if name == "":      # => This means the user didn't type anything
    print("You didn't Enter your name")
else:
    print(f"Hello {name}")

# The Above code will execute only one time

# so we need while loop to execute many times as many as the user didn't type his/her name
while name == "":
    print("You didn't type your name")
    name = input("Enter Your name: ")
print(f"Hello {name}")
