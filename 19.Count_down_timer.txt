# Count down timer Program

import time     # to use sleep() function

time.sleep(2)   #=> it will just wait 2 seconds , then execute the next step
print("TIME'S UP!")

#=============================================================================
# Count down timer Program , with just Seconds
my_time = int(input("Enter the time is seconds: "))
for x in reversed(range(0, my_time)):   # note we used here "reversed" function Or use step "-1" , like this   for x in range(my_time, 0, -1):
    time.sleep(1)
    print(x)

#==============================================================================
# Count down timer Program
my_time1 = int(input("Enter the Time: "))
for x in range(my_time1, -1, -1):
    seconds = x % 60
    minutes = int(x / 60) % 60  # if there wasn't hours, you don't need to put " %60 "
    hours = int(x / 3600)   # If there is "Days" then you should put " % 24 "
    print(f"{hours}:{minutes:02}:{seconds:02}")
    time.sleep(.1)
