# Concession Stand Program

menu = {"pizza": 3.00,
        "nachos": 4.50,
        "popcorn": 6.00,
        "fries": 2.50,
        "chips": 1.00,
        "pretzel": 3.50,
        "soda": 3.00,
        "lemonade": 4.25}

cart = []
total = 0
print("--------- MENU ---------")
for key, value in menu.items():
    print(f"{key:10}: ${value:.2f}")

while True:
    food = input("Select an Item (q to quit): ").lower()
    if food == 'q':
        break
    elif menu.get(food) == None:
        print("This item is not in the list :(")
        continue
    cart.append(food)

print("\n---------------")
print("You cart is: ")
for items in cart:
    print(items, end=" ")

for foodItem in cart:
    total += menu.get(foodItem) # Remember: get() method => Return a value of Key

print(f"---------------\nTotal price is: {total:.2f}")
