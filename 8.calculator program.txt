import math
var1 = float(input("Enter the first number: "))
var2 = float(input("Enter the second number: "))
operator = input("Enter the Operator: ")

multiply = var1*var2
division = var1/var2

if operator == '+':
    print(f"{var1} {operator} {var2} = {var1+var2}")
elif operator == '-':
    print(f"{var1} {operator} {var2} = {var1-var2}")
elif operator == '*':
    print(f"{var1} {operator} {var2} = {round(multiply, 3)}")
elif operator == '/':
    print(f"{var1} {operator} {var2} = {round(division, 3)}")
else:
    print("Numbers or Operator are not valid!")
